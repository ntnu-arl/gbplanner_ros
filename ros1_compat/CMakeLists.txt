cmake_minimum_required(VERSION 3.8)
project(ros1_compat)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(ament_index_cpp REQUIRED)
add_library(ros1_compat INTERFACE)
target_include_directories(ros1_compat INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(ros1_compat INTERFACE rclcpp::rclcpp tf2::tf2 tf2_ros::tf2_ros ament_index_cpp::ament_index_cpp ${geometry_msgs_TARGETS} ${std_srvs_TARGETS})
install(DIRECTORY include/ DESTINATION include)
install(TARGETS ros1_compat EXPORT export_ros1_compat)
ament_export_targets(export_ros1_compat)
ament_export_include_directories(include)
ament_export_dependencies(rclcpp tf2 tf2_ros geometry_msgs std_srvs ament_index_cpp)
ament_package()
