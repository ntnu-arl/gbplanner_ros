cmake_minimum_required(VERSION 3.16)
project(gbplanner_sparse_storage_checks LANGUAGES CXX)
find_package(gbplanner_local REQUIRED)
foreach(check IN ITEMS sparse_observation sparse_free_blocks sparse_box tsdf_query_context)
  add_executable(check_${check} ../check_${check}.cpp)
  target_compile_features(check_${check} PRIVATE cxx_std_17)
  target_compile_options(check_${check} PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(check_${check} PRIVATE gbplanner_local::gbplanner_local)
endforeach()
