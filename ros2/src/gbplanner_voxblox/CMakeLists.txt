cmake_minimum_required(VERSION 3.16)
project(gbplanner_voxblox)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(gbplanner_local REQUIRED)
find_package(voxblox_core REQUIRED)
find_package(voxblox_msgs REQUIRED)
find_package(planner_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "srv/FilePath.srv")
add_library(gbplanner_tsdf SHARED src/tsdf_backend.cpp)
target_include_directories(gbplanner_tsdf PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(gbplanner_tsdf gbplanner_local voxblox_core)
add_executable(voxblox_mapper_node src/voxblox_mapper_node.cpp)
target_link_libraries(voxblox_mapper_node gbplanner_tsdf)
ament_target_dependencies(voxblox_mapper_node rclcpp sensor_msgs nav_msgs geometry_msgs std_srvs tf2_ros visualization_msgs std_msgs voxblox_msgs planner_msgs)
rosidl_get_typesupport_target(map_typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(voxblox_mapper_node "${map_typesupport}")
add_executable(tsdf_terrain_demo src/tsdf_terrain_demo.cpp)
target_link_libraries(tsdf_terrain_demo gbplanner_tsdf)
install(TARGETS gbplanner_tsdf EXPORT tsdfTargets LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(TARGETS voxblox_mapper_node RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS tsdf_terrain_demo RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES layer_SOURCE.json LICENSE.voxblox LICENSE.mesh DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/map_products_demo.py DESTINATION lib/gbplanner_voxblox)
if(BUILD_TESTING)

  ament_target_dependencies(test_distance_products sensor_msgs)

  target_link_libraries(test_mesh_messages gbplanner_tsdf)
  ament_target_dependencies(test_mesh_messages voxblox_msgs)
  ament_add_gtest(test_map_visualization test/test_map_visualization.cpp)
  target_link_libraries(test_map_visualization gbplanner_tsdf)
  ament_target_dependencies(test_map_visualization sensor_msgs visualization_msgs)
  ament_add_gtest(test_incremental_mesh test/test_incremental_mesh.cpp)
  target_link_libraries(test_incremental_mesh gbplanner_tsdf)
  ament_target_dependencies(test_incremental_mesh voxblox_msgs)
  ament_add_gtest(test_transform_queue test/test_transform_queue.cpp)
  target_link_libraries(test_transform_queue gbplanner_tsdf)
  ament_add_gtest(test_icp_refiner test/test_icp_refiner.cpp)

ament_export_targets(tsdfTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(gbplanner_local voxblox_core voxblox_msgs sensor_msgs visualization_msgs rosidl_default_runtime)
