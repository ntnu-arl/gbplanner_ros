cmake_minimum_required(VERSION 3.16)
project(gbplanner_ui)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(planner_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(gbplanner_ui SHARED src/gbplanner_panel.cpp include/gbplanner_ui/gbplanner_panel.hpp)
target_include_directories(gbplanner_ui PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(gbplanner_ui PUBLIC Qt5::Widgets)
ament_target_dependencies(gbplanner_ui PUBLIC rviz_common pluginlib rclcpp std_srvs std_msgs planner_msgs)
target_compile_options(gbplanner_ui PRIVATE -Wall -Wextra -Wpedantic)
pluginlib_export_plugin_description_file(rviz_common plugin_description.xml)
install(TARGETS gbplanner_ui EXPORT uiTargets LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES SOURCE.md LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(panel_runtime test/panel_runtime.cpp)
  target_link_libraries(panel_runtime Qt5::Widgets)
  ament_target_dependencies(panel_runtime rviz_common pluginlib rclcpp std_srvs std_msgs planner_msgs)
  install(TARGETS panel_runtime RUNTIME DESTINATION lib/${PROJECT_NAME})
endif()
ament_export_targets(uiTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(rviz_common pluginlib rclcpp std_srvs std_msgs planner_msgs)
ament_package()
