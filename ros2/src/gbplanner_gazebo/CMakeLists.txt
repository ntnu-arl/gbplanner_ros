cmake_minimum_required(VERSION 3.16)
project(gbplanner_gazebo)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(actuator_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(hector_gazebo_plugins REQUIRED)
find_package(ros_gz_interfaces REQUIRED)
find_package(ros_gz_bridge REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(gbplanner_dynamics REQUIRED)
add_executable(rmf_controller_node src/rmf_controller_node.cpp)
target_link_libraries(rmf_controller_node gbplanner_dynamics::lee_controller Eigen3::Eigen)
ament_target_dependencies(rmf_controller_node rclcpp nav_msgs trajectory_msgs actuator_msgs std_msgs std_srvs ros_gz_interfaces)
install(TARGETS rmf_controller_node RUNTIME DESTINATION lib/${PROJECT_NAME})
add_executable(rmf_imu_adapter_node src/rmf_imu_adapter_node.cpp)
target_link_libraries(rmf_imu_adapter_node Eigen3::Eigen)
ament_target_dependencies(rmf_imu_adapter_node rclcpp rcl_interfaces sensor_msgs std_srvs hector_gazebo_plugins)
install(TARGETS rmf_imu_adapter_node RUNTIME DESTINATION lib/${PROJECT_NAME})
add_executable(managed_bridge_node src/managed_bridge_node.cpp)
ament_target_dependencies(managed_bridge_node rclcpp ros_gz_bridge ros_gz_interfaces
  rosgraph_msgs nav_msgs sensor_msgs actuator_msgs)
install(TARGETS managed_bridge_node RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config worlds models DESTINATION share/${PROJECT_NAME})
install(FILES SOURCE.json DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/fortress_state_node.py scripts/fortress_model.py scripts/fortress_rmf_model.py
  scripts/rmf_mission_node.py scripts/smb_camera_info_node.py
  DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/fortress_smb_sensors.py scripts/fortress_smb_cameras.py
  scripts/fortress_renderer.py DESTINATION lib/${PROJECT_NAME})
ament_package()
