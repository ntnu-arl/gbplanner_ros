cmake_minimum_required(VERSION 3.16)
project(gbplanner_dynamics)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Bullet REQUIRED)
add_library(lee_controller STATIC vendor/rotors_control/src/library/lee_position_controller.cpp)
set_target_properties(lee_controller PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(lee_controller PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/vendor/rotors_control/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(lee_controller Eigen3::Eigen)
add_executable(dynamics_node src/dynamics_node.cpp)
target_include_directories(dynamics_node PRIVATE ${BULLET_INCLUDE_DIRS})
target_link_libraries(dynamics_node lee_controller ${BULLET_LIBRARIES})
ament_target_dependencies(dynamics_node rclcpp nav_msgs sensor_msgs trajectory_msgs std_msgs std_srvs tf2_ros)
install(TARGETS dynamics_node RUNTIME DESTINATION lib/gbplanner_dynamics)
install(TARGETS lee_controller EXPORT export_lee_controller ARCHIVE DESTINATION lib)
install(DIRECTORY include/rotors_control DESTINATION include)
install(FILES vendor/rotors_control/include/rotors_control/parameters.h DESTINATION include/rotors_control)
ament_export_targets(export_lee_controller HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3)
install(PROGRAMS scripts/dynamics_demo.py DESTINATION lib/gbplanner_dynamics)
install(DIRECTORY config launch DESTINATION share/gbplanner_dynamics)
install(FILES SOURCE.json LICENSE DESTINATION share/gbplanner_dynamics)
ament_package()
