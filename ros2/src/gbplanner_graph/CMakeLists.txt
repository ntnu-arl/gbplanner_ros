cmake_minimum_required(VERSION 3.16)
project(gbplanner_graph)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(planner_msgs REQUIRED)
find_package(planner_semantic_msgs REQUIRED)
find_package(kdtree REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Boost REQUIRED)
add_library(gbplanner_graph SHARED src/graph.cpp src/graph_manager.cpp)
target_include_directories(gbplanner_graph PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(gbplanner_graph Eigen3::Eigen)
ament_target_dependencies(gbplanner_graph rclcpp geometry_msgs planner_msgs planner_semantic_msgs kdtree)
add_executable(graph_node src/graph_node.cpp)
target_link_libraries(graph_node gbplanner_graph)
ament_target_dependencies(graph_node rclcpp nav_msgs planner_msgs)
install(TARGETS gbplanner_graph EXPORT graphTargets LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(TARGETS graph_node RUNTIME DESTINATION lib/gbplanner_graph)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/gbplanner_graph)
install(PROGRAMS scripts/graph_demo.py DESTINATION lib/gbplanner_graph)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_graph test/test_graph.cpp)
  target_link_libraries(test_graph gbplanner_graph)
  ament_target_dependencies(test_graph planner_msgs rclcpp)
endif()
ament_export_targets(graphTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp geometry_msgs planner_msgs planner_semantic_msgs kdtree Eigen3 Boost)
ament_package()
