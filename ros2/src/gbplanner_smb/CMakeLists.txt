cmake_minimum_required(VERSION 3.16)
project(gbplanner_smb)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Bullet REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
add_library(smb_vehicle SHARED src/smb_vehicle.cpp src/triangle_mesh.cpp)
target_include_directories(smb_vehicle PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(smb_vehicle SYSTEM PRIVATE ${BULLET_INCLUDE_DIRS})
target_link_libraries(smb_vehicle PUBLIC Eigen3::Eigen PRIVATE ${BULLET_LIBRARIES})
target_compile_options(smb_vehicle PRIVATE -Wall -Wextra -Wpedantic)
add_executable(smb_node src/smb_node.cpp)
target_link_libraries(smb_node smb_vehicle)
ament_target_dependencies(smb_node rclcpp geometry_msgs nav_msgs sensor_msgs std_msgs std_srvs tf2_ros)
target_compile_options(smb_node PRIVATE -Wall -Wextra -Wpedantic)
add_executable(smb_vehicle_demo src/smb_vehicle_demo.cpp)
target_link_libraries(smb_vehicle_demo smb_vehicle)
add_executable(smb_spin_demo src/smb_spin_demo.cpp)
target_link_libraries(smb_spin_demo smb_vehicle)
target_compile_options(smb_spin_demo PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS smb_vehicle EXPORT smbTargets LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(TARGETS smb_node smb_vehicle_demo smb_spin_demo RUNTIME DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/velocity_mux_node.py scripts/joy_teleop_node.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config urdf meshes DESTINATION share/${PROJECT_NAME})
install(FILES SOURCE.json teleop_SOURCE.json LICENSE DESTINATION share/${PROJECT_NAME})
ament_export_targets(smbTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3)
ament_package()
