cmake_minimum_required(VERSION 3.16)
project(gbplanner_tracking)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(planner_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)

add_library(pure_pursuit SHARED src/pure_pursuit.cpp)
target_include_directories(pure_pursuit PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(pure_pursuit PUBLIC Eigen3::Eigen)
target_compile_options(pure_pursuit PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pure_pursuit_node src/pure_pursuit_node.cpp)
target_link_libraries(pure_pursuit_node pure_pursuit ground_maneuver)
ament_target_dependencies(pure_pursuit_node rclcpp rclcpp_action planner_msgs
  geometry_msgs nav_msgs std_msgs std_srvs)
target_compile_options(pure_pursuit_node PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pure_pursuit_demo src/pure_pursuit_demo.cpp)
target_link_libraries(pure_pursuit_demo pure_pursuit)
add_library(ground_maneuver SHARED src/ground_maneuver.cpp)
target_include_directories(ground_maneuver PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(ground_maneuver PUBLIC Eigen3::Eigen)
target_compile_options(ground_maneuver PRIVATE -Wall -Wextra -Wpedantic)
add_executable(ground_maneuver_demo src/ground_maneuver_demo.cpp)
target_link_libraries(ground_maneuver_demo ground_maneuver)
target_compile_options(ground_maneuver_demo PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS pure_pursuit ground_maneuver EXPORT trackingTargets LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(TARGETS pure_pursuit_node pure_pursuit_demo ground_maneuver_demo RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE SOURCE.json DESTINATION share/${PROJECT_NAME})
ament_export_targets(trackingTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3)
ament_package()
